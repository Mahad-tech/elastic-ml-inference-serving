# 11. Monitoring (Run directly in terminal)

"timestamp,replicas,p99_latency,queue_size" | Out-File -FilePath experiment_custom_autoscaler.csv -Encoding utf8

while ($true) {
    $Timestamp = [DateTimeOffset]::Now.ToUnixTimeSeconds()
    
    # 1. Fetch current active ML replica count
    $Replicas = (kubectl get deployment ml-app-deployment -o jsonpath='{.spec.replicas}')
    
    # 2. Query Prometheus for the TRUE 99th percentile dispatcher processing latency
    $LatResponse = Invoke-RestMethod -Uri "http://localhost:9090/api/v1/query?query=histogram_quantile(0.99,+sum(rate(dispatcher_response_time_seconds_bucket[1m]))+by+(le))"
    $Latency = "0.0"
    if ($null -ne $LatResponse.data.result -and $LatResponse.data.result.Count -gt 0) {
        $Latency = $LatResponse.data.result[0].value[1]
    }
    
    # 3. Query Prometheus for the real-time queue size gauge
    $QueueResponse = Invoke-RestMethod -Uri "http://localhost:9090/api/v1/query?query=dispatcher_queue_size"
    $Queue = "0"
    if ($null -ne $QueueResponse.data.result -and $QueueResponse.data.result.Count -gt 0) {
        $Queue = $QueueResponse.data.result[0].value[1]
    }
    
    # Output progress cleanly
    $LogLine = "$Timestamp,$Replicas,$Latency,$Queue"
    Write-Output $LogLine
    $LogLine | Out-File -FilePath experiment_custom_autoscaler.csv -Append -Encoding utf8
    
    Start-Sleep -Seconds 10
}
