  ELASTIC ML INFERENCE SERVING — LINUX STARTUP GUIDE
  Includes: Deployment, Grafana, HPA Experiments


!! Activate your Python virtual environment !!
cd ~/elastic-ml-inference-serving
source .venv/bin/activate


TERMINAL 1 — SETUP & DEPLOY (Main Control Terminal)


# 1. Start Minikube with enough resources
minikube delete
minikube start --cpus=4 --memory=7000m --driver=docker

# 2. Point shell to Minikube's internal Docker registry
eval $(minikube docker-env)
docker ps

# 3. Install Prometheus Operator CRDs
kubectl apply --server-side -f https://raw.githubusercontent.com/prometheus-operator/prometheus-operator/main/bundle.yaml

# 4. Create ServiceAccount and ClusterRoleBinding for autoscaler
kubectl apply -f - << 'EOF'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: python-client-sa
  namespace: default
EOF

kubectl apply -f - << 'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: python-client-sa-admin-binding
subjects:
- kind: ServiceAccount
  name: python-client-sa
  namespace: default
roleRef:
  kind: ClusterRole
  name: cluster-admin
  apiGroup: rbac.authorization.k8s.io
EOF

# 5. Build all 4 container images
docker build -t ml-app:latest -f ml-app/Dockerfile .
docker build -t dispatcher:latest -f dispatcher/Dockerfile .
docker build -t autoscaler:latest -f custom_autoscaler/Dockerfile .
docker build -t load-tester:local -f load_tester_config/Dockerfile.tester .

# 6. Deploy all components
kubectl apply -f manifests/ml-app-deployment.yaml
kubectl apply -f manifests/prometheus-rbac.yaml
kubectl apply -f manifests/dispatcher-deployment.yaml
kubectl apply -f manifests/prometheus-instance.yaml
kubectl apply -f manifests/custom-autoscaler-deployment.yaml
kubectl apply -f manifests/grafana.yaml

# 7. Install kube-state-metrics (needed for Replicas panel in Grafana)
kubectl apply -f https://raw.githubusercontent.com/kubernetes/kube-state-metrics/main/examples/standard/cluster-role.yaml
kubectl apply -f https://raw.githubusercontent.com/kubernetes/kube-state-metrics/main/examples/standard/cluster-role-binding.yaml
kubectl apply -f https://raw.githubusercontent.com/kubernetes/kube-state-metrics/main/examples/standard/service-account.yaml
kubectl apply -f https://raw.githubusercontent.com/kubernetes/kube-state-metrics/main/examples/standard/deployment.yaml
kubectl apply -f https://raw.githubusercontent.com/kubernetes/kube-state-metrics/main/examples/standard/service.yaml

# 8. Add ServiceMonitor for kube-state-metrics
kubectl apply -f - << 'EOF'
apiVersion: monitoring.coreos.com/v1
kind: ServiceMonitor
metadata:
  name: kube-state-metrics
  namespace: default
  labels:
    release: prometheus
spec:
  namespaceSelector:
    matchNames:
    - kube-system
  selector:
    matchLabels:
      app.kubernetes.io/name: kube-state-metrics
  endpoints:
  - port: http
EOF

# 9. Wait for Prometheus Operator to be ready
kubectl wait --for=condition=Ready pod -l app.kubernetes.io/name=prometheus-operator --timeout=120s

# 10. Watch all pods come up (Ctrl+C once all show Running)
kubectl get pods -w

# 11. Verify custom autoscaler is deployed
kubectl get deployment custom-autoscaler-deployment

# 12. Start load tester
kubectl delete job load-tester-job --ignore-not-found
kubectl apply -f manifests/load-tester-job.yaml

# 13. Watch load tester logs
kubectl logs -l job-name=load-tester-job -f

TERMINAL 4 — AUTOSCALER LOGS + METRICS COLLECTION


cd ~/elastic-ml-inference-serving
source .venv/bin/activate

# Option A: Watch autoscaler logs
kubectl logs -l app=custom-autoscaler -f

# Option B: Collect metrics to CSV (use during experiments)
echo "timestamp,replicas,p99_latency,queue_size" > experiment_custom_autoscaler.csv

while true; do
    TIMESTAMP=$(date +%s)
    REPLICAS=$(kubectl get deployment ml-app-deployment -o jsonpath='{.spec.replicas}')
    LATENCY=$(curl -s "http://localhost:9090/api/v1/query?query=histogram_quantile(0.99,sum(rate(dispatcher_response_time_seconds_bucket[1m]))by(le))" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d['data']['result'][0]['value'][1] if d['data']['result'] else '0.0')" 2>/dev/null)
    QUEUE=$(curl -s "http://localhost:9090/api/v1/query?query=dispatcher_queue_size" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d['data']['result'][0]['value'][1] if d['data']['result'] else '0')" 2>/dev/null)
    echo "$TIMESTAMP,$REPLICAS,$LATENCY,$QUEUE" | tee -a experiment_custom_autoscaler.csv
    sleep 10
done



TERMINAL 5 — HPA EXPERIMENTS


cd ~/elastic-ml-inference-serving
source .venv/bin/activate

# ---- EXPERIMENT 2: HPA 70% CPU ----

# Stop custom autoscaler
kubectl scale deployment custom-autoscaler-deployment --replicas=0

# Reset ml-app to 1 replica
kubectl scale deployment ml-app-deployment --replicas=1

# Apply HPA 70%
kubectl apply -f manifests/hpa-70.yaml
kubectl get hpa -w
# Wait 30 seconds for HPA to initialize then Ctrl+C

# Run load tester
kubectl delete job load-tester-job --ignore-not-found
kubectl apply -f manifests/load-tester-job.yaml
kubectl logs -l job-name=load-tester-job -f
# Take Grafana screenshot when load test is running

# ---- EXPERIMENT 3: HPA 90% CPU ----

# Remove HPA 70%
kubectl delete hpa --all

# Reset ml-app to 1 replica
kubectl scale deployment ml-app-deployment --replicas=1

# Apply HPA 90%
kubectl apply -f manifests/hpa-90.yaml
kubectl get hpa -w
# Wait 30 seconds for HPA to initialize then Ctrl+C

# Run load tester
kubectl delete job load-tester-job --ignore-not-found
kubectl apply -f manifests/load-tester-job.yaml
